<!-- ADD THIS JAVASCRIPT -->

<script>
function cycleReveal(card) {

  if (!card.classList.contains("reveal-hebrew")) {
    card.classList.add("reveal-hebrew");
    return;
  }

  if (!card.classList.contains("reveal-english")) {
    card.classList.add("reveal-english");
    return;
  }

  if (!card.classList.contains("reveal-translit")) {
    card.classList.add("reveal-translit");
    return;
  }

  card.classList.remove(
    "reveal-hebrew",
    "reveal-english",
    "reveal-translit"
  );
}
</script>
